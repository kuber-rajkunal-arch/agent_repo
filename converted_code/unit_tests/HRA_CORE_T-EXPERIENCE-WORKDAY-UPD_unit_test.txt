-- test: check_if_curr_ind_is_updated_to_N
-- Verifies that for candidates with multiple MOST_RCNT_IND values, the CURR_IND for the record with MOST_RCNT_IND = 'no' is updated to 'N'.
SELECT IF(
  (SELECT COUNT(*) FROM `gcp-project-id.HRA_CORE_V.EXPERIENCE` WHERE CANDIDATE_ID IN (SELECT CANDIDATE_ID FROM `gcp-project-id.HRA_CORE_V.EXPERIENCE` GROUP BY CANDIDATE_ID HAVING COUNT(DISTINCT MOST_RCNT_IND) > 1) AND MOST_RCNT_IND = 'no' AND CURR_IND = 'Y') = 0,
  'PASS', 'FAIL') AS result;

-- test: check_if_eff_end_dttm_is_populated
-- Verifies that EFF_END_DTTM is populated for the records that were updated.
SELECT IF(

  'PASS', 'FAIL') AS result;
